	Procedimiento CargarLecturas(var lecturas: ARREGLO[1..N] de real, N: entero, sensorElegido: entero) es
		Proceso
			ABRIR E/(archLecturas);
  			leer(archLecturas, reg);

		    MIENTRAS NFDA(archLecturas) HACER
        		SI reg.sensor = sensorElegido ENTONCES
            		SI reg.nro >= 1 Y reg.nro <= N ENTONCES
                		lecturas[reg.nro] := reg.valor;
            		FIN_SI
        		FIN_SI

      		  LEER(archLecturas, reg);
    		FIN_MIENTRAS

  			CERRAR(archLecturas);
	fin_procedimiento

	Procedimiento Estadisticas(lecturas: ARREGLO[1..N] de real, N: entero; maximo: real; posMaximo: entero; minimo: real; posMinimo: entero; promedio: real) es
		Proceso
    		suma := 0;

    		maximo := lecturas[1];
    		minimo := lecturas[1];

   			posMaximo := 1;
    		posMinimo := 1;

    		PARA i := 1 HASTA N HACER
        		suma := suma + lecturas[i];

        		SI lecturas[i] > maximo ENTONCES
            		maximo := lecturas[i];
            		posMaximo := i;
        		FIN_SI

        		SI lecturas[i] < minimo ENTONCES
            		minimo := lecturas[i];
            		posMinimo := i;
        		FIN_SI
    		FIN_PARA

    		promedio := suma / N;
	fin_procedimiento

	Procedimiento DetectarRacha(lecturas: ARREGLO[1..N] de real; N: entero; umbral: real; longitudMinima: entero; encontrada: logico; posInicio: entero);
		Proceso
    		contador := 0;
    		encontrada := FALSO;
    		posInicio := 0;

    		i := 1;

    		MIENTRAS i <= N Y encontrada = FALSO HACER
        		SI lecturas[i] > umbral ENTONCES
           			contador := contador + 1;

            		SI contador = 1 ENTONCES
                		inicio := i;
            		FIN_SI

            		SI contador >= longitudMinima ENTONCES
                		encontrada := VERDADERO;
                		posInicio := inicio;
            		FIN_SI
        		SINO
           			contador := 0;
        		FIN_SI
       			i := i + 1;
	    	FIN_MIENTRAS
	fin_procedimiento

	Procedimiento BuscarUmbral(lecturas: ARREGLO[1..N] de real; N: entero; umbral: real; posicion: entero);
	Proceso
    	posicion := 0;
    	i := 1;

    	MIENTRAS i <= N Y posicion = 0 HACER
        	SI lecturas[i] > umbral ENTONCES
            	posicion := i;
        	SINO
            	i := i + 1;
        	FIN_SI
    	FIN_MIENTRAS
	fin_procedimiento

Proceso
